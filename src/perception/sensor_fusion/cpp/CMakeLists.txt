cmake_minimum_required(VERSION 3.16)
project(a1_fusion_cpp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(Eigen3 REQUIRED)
add_library(a1_core STATIC src/core.cpp)
target_include_directories(a1_core PUBLIC include)
target_link_libraries(a1_core PUBLIC Eigen3::Eigen)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests a1_core)
add_test(NAME core_tests COMMAND core_tests)
add_library(a1_association STATIC src/association.cpp)
target_include_directories(a1_association PUBLIC include)
target_link_libraries(a1_association PUBLIC Eigen3::Eigen)
add_executable(association_tests tests/association_tests.cpp)
target_link_libraries(association_tests a1_association)
add_test(NAME association_tests COMMAND association_tests)
add_executable(p5_cli src/p5_cli.cpp)
target_link_libraries(p5_cli a1_core)
add_executable(replay_p5_csv src/replay_p5_csv.cpp)
target_link_libraries(replay_p5_csv a1_core)
option(BUILD_ROS "Build ROS 2 sensor subscribers" ON)
option(BUILD_TORCH_CAMERA "Build C++ TorchScript camera inference" OFF)

  rosidl_generate_interfaces(${PROJECT_NAME} "msg/A2Features.msg" "msg/BrakeEstimate.msg" "msg/ClusterTrack.msg" "msg/Association.msg" "msg/EgoMotion.msg" DEPENDENCIES std_msgs builtin_interfaces)
  rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
  add_executable(lidar_domain_bridge src/lidar_domain_bridge.cpp)
  ament_target_dependencies(lidar_domain_bridge rclcpp sensor_msgs)
  install(TARGETS lidar_domain_bridge DESTINATION lib/${PROJECT_NAME})

  target_include_directories(fusion_node PRIVATE ${PCL_INCLUDE_DIRS})

  install(TARGETS fusion_node p5_cli replay_p5_csv DESTINATION lib/${PROJECT_NAME})

  find_package(cv_bridge REQUIRED)
  find_package(OpenCV REQUIRED)
  set(PATCHWORK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/patchwork-plusplus/cpp")

  target_include_directories(a1_patchworkpp PUBLIC "${PATCHWORK_ROOT}/patchworkpp/include" "${PATCHWORK_ROOT}/common/include")
  target_link_libraries(a1_patchworkpp PUBLIC Eigen3::Eigen)
  add_library(a1_map_surfaces STATIC src/map_surfaces.cpp)
  target_include_directories(a1_map_surfaces PUBLIC include ${PCL_INCLUDE_DIRS})

  target_link_libraries(map_surface_export a1_map_surfaces mola::mp2p_icp_map mola::mola_metric_maps dl)

    add_executable(a2_camera_node src/a2_camera_node.cpp)
    # Current host LibTorch requires C++20; the sensor/P5 core remains C++17.

    target_link_libraries(a2_camera_node "${TORCH_LIBRARIES}" ${OpenCV_LIBRARIES} "${typesupport}")
    separate_arguments(TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
    target_compile_options(a2_camera_node PRIVATE ${TORCH_FLAGS})
    ament_target_dependencies(a2_camera_node rclcpp sensor_msgs cv_bridge)
